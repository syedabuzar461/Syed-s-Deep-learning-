import tensorflow as tf
from tensorflow.keras import layers, Model
from tensorflow.keras.applications import DenseNet121

IMG_SIZE = 256
NUM_CLASSES = 12

# Input
inputs = layers.Input(
    shape=(IMG_SIZE, IMG_SIZE, 3)
)

# DenseNet backbone
backbone = DenseNet121(
    weights="imagenet",
    include_top=False,
    input_tensor=inputs
)

# Freeze early DenseNet layers
for layer in backbone.layers[:200]:
    layer.trainable = False

x = backbone.output

# Decoder
x = layers.Conv2D(
    256, 3,
    padding="same",
    activation="relu"
)(x)

x = layers.BatchNormalization()(x)

x = layers.UpSampling2D((2, 2))(x)

x = layers.Conv2D(
    128, 3,
    padding="same",
    activation="relu"
)(x)

x = layers.UpSampling2D((2, 2))(x)

x = layers.Conv2D(
    64, 3,
    padding="same",
    activation="relu"
)(x)

x = layers.UpSampling2D((2, 2))(x)

x = layers.Conv2D(
    32, 3,
    padding="same",
    activation="relu"
)(x)

x = layers.UpSampling2D((2, 2))(x)

# Pixel-wise prediction
outputs = layers.Conv2D(
    NUM_CLASSES,
    1,
    activation="softmax"
)(x)

model = Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=0.0001),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

model.summary()
