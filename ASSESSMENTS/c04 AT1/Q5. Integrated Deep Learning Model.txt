import tensorflow as tf
from tensorflow.keras import layers, Model
from tensorflow.keras.applications import EfficientNetB0

IMG_SIZE = 224
FRAME_COUNT = 20
VOCAB_SIZE = 10000
EMBED_DIM = 256
LSTM_UNITS = 512

# ---------------- CNN FEATURE EXTRACTOR ----------------

cnn = EfficientNetB0(
    weights="imagenet",
    include_top=False,
    pooling="avg",
    input_shape=(IMG_SIZE, IMG_SIZE, 3)
)

# Freeze pre-trained CNN
cnn.trainable = False

frame_input = layers.Input(
    shape=(IMG_SIZE, IMG_SIZE, 3)
)

frame_features = cnn(frame_input)

feature_extractor = Model(
    frame_input,
    frame_features
)

# ---------------- VIDEO INPUT ----------------

video_input = layers.Input(
    shape=(
        FRAME_COUNT,
        IMG_SIZE,
        IMG_SIZE,
        3
    ),
    name="video_input"
)

# Extract features from every frame
features = layers.TimeDistributed(
    feature_extractor
)(video_input)

# ---------------- LSTM ENCODER ----------------

encoder_lstm = layers.LSTM(
    LSTM_UNITS,
    return_state=True
)

_, encoder_h, encoder_c = encoder_lstm(
    features
)

# ---------------- CAPTION DECODER ----------------

caption_input = layers.Input(
    shape=(None,),
    name="caption_input"
)

embedding = layers.Embedding(
    VOCAB_SIZE,
    EMBED_DIM,
    mask_zero=True
)(caption_input)

decoder_lstm = layers.LSTM(
    LSTM_UNITS,
    return_sequences=True,
    return_state=True
)

decoder_output, _, _ = decoder_lstm(
    embedding,
    initial_state=[
        encoder_h,
        encoder_c
    ]
)

# Predict next word
word_output = layers.Dense(
    VOCAB_SIZE,
    activation="softmax"
)(decoder_output)

# Complete model
model = Model(
    [video_input, caption_input],
    word_output
)

model.compile(
    optimizer=tf.keras.optimizers.Adam(
        learning_rate=0.0001
    ),
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

model.summary()

# ---------------- CAPTION GENERATION ----------------

def generate_caption(
    model,
    video,
    tokenizer,
    max_length=30
):

    start_token = tokenizer.word_index["<start>"]
    end_token = tokenizer.word_index["<end>"]

    generated = [start_token]

    for _ in range(max_length):

        decoder_input = tf.expand_dims(
            generated,
            axis=0
        )

        prediction = model.predict(
            [
                tf.expand_dims(video, axis=0),
                decoder_input
            ],
            verbose=0
        )

        next_word = int(
            tf.argmax(
                prediction[0, -1]
            )
        )

        generated.append(next_word)

        if next_word == end_token:
            break

    words = []

    for token in generated:

        if token in tokenizer.index_word:

            word = tokenizer.index_word[token]

            if word not in ["<start>", "<end>"]:
                words.append(word)

    return " ".join(words)
